FoamFile
{
    format      ascii;
    class       volVectorField;
    location    "0";
    object      U;
}
// * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * * //

dimensions      [0 1 -1 0 0 0 0];

internalField   uniform (0 0 0);

boundaryField
{
    #includeEtc "caseDicts/setConstraintTypes"

    staticWalls
    {
        type            noSlip;
    }

    // Walls of the rotating cellZone: fluid sticks to the moving mesh,
    // which is what drives the mixing
    movingWalls
    {
        type            movingWallVelocity;
        value           uniform (0 0 0);
    }

    // Leftover original interface faces. movingWallVelocity imposes the local
    // mesh velocity, so it is correct for both the rotor-side (moving) and
    // stator-side (stationary) remainder faces.
    nonCouple
    {
        type            movingWallVelocity;
        value           uniform (0 0 0);
    }
}
